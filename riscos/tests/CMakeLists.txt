# Phase 0 smoke tests for the RISC OS toolchain. Build with riscos/scripts/build-tests.sh.
cmake_minimum_required(VERSION 3.25)
project(RiscosSmokeTests C CXX)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_executable(hello hello.cpp)
add_executable(unaligned unaligned.c)
# The probe uses halfword instructions on purpose; the toolchain targets ARMv3M.
if(RISCOS)
    target_compile_options(unaligned PRIVATE -march=armv4)
endif()
add_executable(dirprobe dirprobe.c)
add_executable(bigalloc bigalloc.c)
add_executable(readspeed readspeed.c)

find_package(SDL REQUIRED)
add_executable(sdltest sdltest.cpp)
add_executable(paltest paltest.cpp)
add_executable(audiotest audiotest.cpp)
add_executable(modetest modetest.cpp)
target_include_directories(modetest PRIVATE ${SDL_INCLUDE_DIR})
target_link_libraries(modetest PRIVATE ${SDL_LIBRARY})
target_include_directories(audiotest PRIVATE ${SDL_INCLUDE_DIR})
target_link_libraries(audiotest PRIVATE ${SDL_LIBRARY})
target_include_directories(paltest PRIVATE ${SDL_INCLUDE_DIR})
target_link_libraries(paltest PRIVATE ${SDL_LIBRARY})
target_include_directories(sdltest PRIVATE ${SDL_INCLUDE_DIR})
target_link_libraries(sdltest PRIVATE ${SDL_LIBRARY})

if(RISCOS)
    foreach(t hello unaligned sdltest dirprobe paltest audiotest bigalloc readspeed modetest)
        riscos_make_aif(${t})
    endforeach()
endif()
